cmake_minimum_required(VERSION 3.20)
project(oaths_and_omens_engine CXX)

# See engine/PLAN.md §1.2: the core must never depend on Python. `bindings/` is
# the only place pybind11 may appear, and it lives behind this option so that
# -DOO_BUILD_PYTHON=OFF produces a build with no Python involvement at all -
# at build time or run time. Both configurations should be built in CI so the
# dependency cannot creep back in.
option(OO_BUILD_PYTHON "Build the transitional pybind11 module (deleted at M8)" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

if(MSVC)
  add_compile_options(/W4 /permissive-)
  # Release: full optimization + AVX2 + link-time codegen. /fp:fast is safe here
  # because no float value is ever compared for parity - random() is built from
  # exact integer arithmetic on powers of two (see rng.hpp), and the evaluation
  # heuristics are ranked, not equality-tested.
  add_compile_options($<$<CONFIG:Release>:/O2>
                      $<$<CONFIG:Release>:/fp:fast>
                      $<$<CONFIG:Release>:/arch:AVX2>
                      $<$<CONFIG:Release>:/GL>)
  add_link_options($<$<CONFIG:Release>:/LTCG>)
  # Debug keeps assertions on - the bounds-checked, invariant-asserting build is
  # what makes the parity work tractable (PLAN.md §4.5).
  add_compile_options($<$<CONFIG:Debug>:/Od> $<$<CONFIG:Debug>:/RTC1>)
else()
  add_compile_options(-Wall -Wextra)
  add_compile_options($<$<CONFIG:Release>:-O3>)
endif()

# --- core library -----------------------------------------------------------
# Sources get added here as modules land. rng.hpp stays header-only (it is small
# and hot enough to want inlined).
add_library(oo_core STATIC
  src/grid.cpp
  src/state.cpp
  src/state_io.cpp
  src/terrain.cpp
  src/collect.cpp
  src/buy.cpp
  src/movement.cpp
  src/battle.cpp
  src/turn.cpp
  src/setup.cpp
  src/placement.cpp
)
target_include_directories(oo_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

# --- transitional pybind11 module (PLAN.md 1.2, 5) --------------------------
# Deleted at M8. Nothing in oo_core depends on this; it depends on oo_core.
if(OO_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(
    COMMAND "${Python_EXECUTABLE}" -c "import pybind11; print(pybind11.get_cmake_dir())"
    OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(oo_engine bindings/module.cpp)
  target_link_libraries(oo_engine PRIVATE oo_core)
  # Put the .pyd beside the engine/ shim package so `import engine` finds it
  # without any PYTHONPATH juggling.
  set_target_properties(oo_engine PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    LIBRARY_OUTPUT_DIRECTORY_RELEASE ${CMAKE_CURRENT_SOURCE_DIR}
    LIBRARY_OUTPUT_DIRECTORY_DEBUG ${CMAKE_CURRENT_SOURCE_DIR})
endif()

# The golden trace is regenerated by tools/dump_rng_reference.py. It is checked
# in, so running the tests needs no Python (PLAN.md §1.2) - Python is only

         COMMAND test_rng ${CMAKE_CURRENT_SOURCE_DIR}/tests/data/rng_golden.txt)
